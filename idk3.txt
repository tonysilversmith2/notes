import pandas as pd 

def transform_product_sales_format(sales_file):

    wide_df=pd.read_csv(sales_file)

    id_vars=['product_id', 'product_name', 'category']

    long_df= pd.melt(
        wide_df,
        id_vars=id_vars,
        var_name='region',
        value_name='revenue'
    )
    reshaped_wide=pd.pivot_table(
        long_df,
        index=id_vars,
        columns='region',
        values='revenue',
        aggfunc='sum',
        fill_value=0
    )

    reshaped_wide.columns.name=None
    reshaped_wide= reshaped_wide.reset_index()

    return long_df, reshaped_wide

import pandas as pd

# 1. Create an in-memory SQLite database
conn = sqlite3.connect(":memory:")
cursor = conn.cursor()

# 2. Create the books and sales tables
cursor.execute(
    """
    CREATE TABLE books (
        book_id INTEGER PRIMARY KEY,
        title TEXT NOT NULL,
        genre TEXT NOT NULL,
        stock INTEGER NOT NULL,
        price REAL NOT NULL
    )
"""
)

cursor.execute(
    """
    CREATE TABLE sales (
        sale_id INTEGER PRIMARY KEY,
        book_id INTEGER,
        quantity INTEGER NOT NULL,
        FOREIGN KEY (book_id) REFERENCES books(book_id)
    )
"""
)

# 3. Main SQL query for Revenue by Genre
sql_query = """
    SELECT 
        b.genre,
        SUM(s.quantity * b.price) AS total_revenue,
        SUM(s.quantity) AS total_sold
    FROM books b
    JOIN sales s ON b.book_id = s.book_id
    GROUP BY b.genre
    ORDER BY total_revenue DESC
"""

# Execute and print Part 1
print("Revenue by Genre:")
df_revenue = pd.read_sql_query(sql_query, conn)
print(df_revenue.to_string())

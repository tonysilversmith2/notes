import pandas as pd 
def generate_customer_spending_report(customers_file,orders_file, order_items_file):

    df_cust=pd.read_csv(customers_file)
    df_ord=pd.read_csv(orders_file)
    df_items=pd.read_csv(order_items_file)
    
    df_items['line_total']= df_items['quantity'] * df_items['unit_price']

    
    df_merged= df_cust.merge( df_ord ,on='customer_id', how='left')
    df_merged= df_merged.merge( df_items ,on='order_id', how='left')



    summary= df_merged.groupby( ["customer_id",'customer_name']).agg(
        order_count=('order_id', 'nunique'),
        total_spend=('line_total','sum')
    ).reset_index()

    

    
    summary['order_count']= summary['order_count'].fillna(0).astype(int)
    summary['total_spend']= summary['total_spend'].fillna(0.0).astype(float)
    
    summary= summary.sort_values(by='total_spend', ascending=False).reset_index(drop=True)

    return summary[["customer_id",'customer_name','order_count','total_spend']]

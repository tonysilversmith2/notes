import pandas as pd 

def combine_department_reports(csv_files):

    all_dfs=[]

    for file in csv_files:
        df=pd.read_csv(file)

        df['month']=file.replace('.csv','')

        all_dfs.append(df)

    combined_df=pd.concat(all_dfs, ignore_index=True)

    combined_df=combined_df.sort_values(by=['dept_id','month'], ascending=[True, True])

    return combined_df.reset_index(drop=True)
